[31m+  * 解析数据绑定表达式（`{ path: '/a/b' }`）为 dataModel 中的实际值[39m

[31m+ // ─── 流式响应事件监听（StreamStart/StreamDelta/StreamEnd）────────────────────[39m
[31m+[39m
[31m+ /**[39m
[31m+  * 按 id 同时更新 `messages` 与 `sessionMessagesCache` 中的消息[39m

[31m+   }[39m
[31m+ }[39m

[31m+ /**[39m
[31m+  * 处理一条流式事件，驱动打字机效果[39m

[31m+  * - `start`：创建一条空助手消息并推入消息列表，开始累积[39m

[31m+   if (sessionId !== currentSessionId.value) return[39m

[31m+       used: payload.used,[39m
[31m+       limit: payload.limit,[39m
[31m+       percent: payload.percent,[39m
[31m+     }[39m

[31m+   }[39m

[31m+     latestRetry.value = {[39m
[31m+       attempt: payload.attempt,[39m
[31m+       maxRetries: payload.maxRetries,[39m
[31m+       backoffMs: payload.backoffMs,[39m
[31m+       error: payload.error,[39m
[31m+       deadlineAt: Date.now() + payload.backoffMs,[39m
[31m+     }[39m
[31m+     return[39m
[31m+   }[39m
[31m+   if (payload.type === 'error') {[39m
[31m+     latestErrorInfo.value = {[39m
[31m+       code: payload.code,[39m
[31m+       retryable: payload.retryable,[39m
[31m+       retryAfterMs: payload.retryAfterMs,[39m

[31m+   // - tool_delta：bash/powershell 增量输出 → 归并到正在执行/刚完成的工具段内容区[39m

[31m+     const turnId = currentTurnId.value[39m

[31m+       if (payload.type === 'tool_delta') {[39m
[31m+         applyTurnEvent(sessionId, {[39m
[31m+           type: 'tool_delta',[39m
[31m+           turnId,[39m
[31m+           toolCallId: payload.tool_call_id ?? '',[39m
[31m+           delta: payload.delta ?? '',[39m
[31m+         })[39m
[31m+       } else {[39m
[31m+         applyTurnEvent(sessionId, {[39m
[31m+           type: 'file_change',[39m
[31m+           turnId,[39m
[31m+           toolCallId: payload.tool_call_id ?? '',[39m
[31m+           path: payload.path ?? '',[39m
[31m+           kind: payload.kind ?? '',[39m
[31m+           hunks: payload.hunks ?? [],[39m

[31m+     return[39m
[31m+   }[39m
[31m+[39m
[31m+   if (payload.type === 'start') {[39m
[31m+     // A5 收尾：流恢复（新流开始）→ 重试状态行消散（若曾重试）[39m
[31m+     latestRetry.value = null[39m
[31m+     // 新流开始：创建空助手消息承载后续增量（messages 仍维护，供 lastAssistantHas* 等消费）[39m
[31m+     const msg: AgentMessage = {[39m
[31m+       id: nextAssistantId(),[39m
[31m+       role: 'assistant',[39m
[31m+       content: '',[39m
[31m+       timestamp: Date.now(),[39m
[31m+     }[39m

[31m+[39m
[31m+   if (payload.type === 'delta') {[39m

[31m+[39m

[31m+   if (payload.type === 'tool_call_start') {[39m
[31m+     if (turnId) {[39m
[31m+       applyTurnEvent(sessionId, {[39m
[31m+         type: 'action_start',[39m
[31m+         turnId,[39m
[31m+         id: `tool-${payload.toolCallId}`,[39m
[31m+         action: 'tool',[39m

[31m+         refId: payload.toolCallId,[39m

[31m+       // H7 P0.5：事件顶层 status/error 权威（后端真实执行结果）；解析推断仅作旧事件回退[39m

[31m+         turnId,[39m
[31m+         refId: payload.toolCallId,[39m
[31m+         status,[39m
[31m+         result: parsed.result,[39m

[31m+       })[39m

[31m+   // 段收尾与结果层固化不在此处——turn_end 由 prompt 的 send_message 返回路径统一发送，[39m
[31m+   // 避免「事件先到、返回值后到」两次 turn_end 造成结果层重复段[39m

[31m+   if (turnId) applyTurnEvent(sessionId, { type: 'actions_settle', turnId })[39m

[31m+   } catch (err) {[39m
[31m+     console.error(`[useAgentRust] listen ${eventName} 失败:`, err)[39m

[31m+ > = {[39m

[31m+ /**[39m

[31m+  * 与实时 emitPlanStepsIntoFlow 同构：逐项建段（order/description 透传），再按状态收口——[39m

[31m+ /** 2026-09-02 计划步骤并入过程层：plan_update 步骤状态流转 → 驱动 plan 动作段（in_progress 激活 / 完成收口） */[39m

[31m+ async function setupPlanListener(sessionId: string): Promise<void> {[39m

[31m+       usePlanStore().createPlan([39m
[31m+         e.payload.items.map((p) => ({[39m
[31m+           id: p.id,[39m
[31m+           title: p.title,[39m
[31m+           status: p.status,[39m
[31m+           description: p.description ?? undefined,[39m
[31m+           order: p.order,[39m
[31m+           output: p.output ?? null,[39m
[31m+           reason: p.reason ?? null,[39m
[31m+         })),[39m
[31m+       )[39m
[31m+       // 2026-09-04 待办项升级：对齐 turnModel —— 删除旧 plan 段，按新 items 全量重建[39m
[31m+       const turnId = currentTurnId.value[39m
[31m+       if (turnId) {[39m
[31m+         applyTurnEvent(sessionId, {[39m
[31m+           type: 'plan_replanned',[39m
[31m+           turnId,[39m
[31m+           items: e.payload.items.map((it) => ({[39m
[31m+             stepId: it.id,[39m
[31m+             order: it.order,[39m
[31m+             // Rust in_progress → 前端 running（重规划后新激活的步骤进入运行态展开）[39m
[31m+             status: it.status === 'in_progress' ? 'running' : it.status,[39m
[31m+             title: it.title,[39m
[31m+             description: it.description ?? undefined,[39m
[31m+             reason: it.reason ?? undefined,[39m
[31m+             result: it.output ?? undefined,[39m
