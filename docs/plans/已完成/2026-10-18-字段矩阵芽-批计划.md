`buildMatrixModel.ts` 核心逻辑：

```ts
export function buildMatrixModel(sources: FieldMatrixSources): FieldMatrixModel {
  const traitMap = new Map(sources.traits.map(trait => [trait.id, trait]))
  const entities = sources.schemas.map(schema => buildEntity(schema, traitMap, sources.userFields(schema.type)))
  const views = sources.views.map(view => ({
    id: view.id, label: view.label, icon: view.icon,
    pluginId: view.pluginId ?? '',
    toolbar: view.toolbar ?? 'bar',
    entityTypes: sources.schemas.filter(schema => schema.pluginId === view.pluginId).map(schema => schema.type),
  }))
  const instances = entities.reduce((sum, entity) => sum + entity.fields.length, 0)
  return {
    entities,
    traits: sources.traits.map(trait => ({ id: trait.id, label: trait.label, usage: sources.schemas.filter(schema => schema._traitIds.includes(trait.id)).length, fields: trait.fields })),
    facets: sources.facets.map(facet => ({ id: facet.id, label: facet.label, hostTypes: facet.hostTypes, fields: facet.fields.map(field => toMatrixField(facet.id, field, 'facet')) })),
    relations: sources.relations.map(relation => ({
      type: relation.type, label: relation.label,
      sourceTypes: relation.sourceTypes, targetTypes: relation.targetTypes,
      inverseType: relation.inverseType, inverseLabel: relation.inverseLabel,
      symmetric: relation.symmetric, domain: relation.domain,
      propertyCount: relation.properties?.length ?? 0,
    })),
    views,
    stats: {
      buds: new Set(sources.schemas.map(schema => schema.pluginId).filter(Boolean)).size,
      entities: sources.schemas.length,
      instances,
      keys: new Set(entities.flatMap(entity => entity.fields.map(field => field.key))).size,
      traits: sources.traits.length,
      facets: sources.facets.length,
    },
  }
}
```

`buildEntity` 的合并骨架：

```ts
function buildEntity(
  schema: EntitySchemaV2,
  traitMap: Map<string, TraitDefinition>,
  userFields: FieldSchema[],
): MatrixEntity {
  const fields: MatrixField[] = []
  const sources = new Map<string, string>()
  for (const ref of schema.traits) {
    const trait = traitMap.get(ref.traitId)
    if (!trait) continue
    for (const field of trait.fields) {
      const override = ref.fieldOverrides?.[field.key]
      const merged = override ? { ...field, ...override } : { ...field }
      if (sources.has(merged.key)) continue
      const source = `trait:${ref.traitId}${override ? '+override' : ''}`
      sources.set(merged.key, source)
      fields.push(toMatrixField(schema.type, merged, source))
    }
  }
  for (const field of schema.ownFields) {
    const index = fields.findIndex(item => item.key === field.key)
    const source = index >= 0 ? `own+replace(${sources.get(field.key)})` : 'own'
    if (index >= 0) fields[index] = toMatrixField(schema.type, field, source)
    else {
      fields.push(toMatrixField(schema.type, field, source))
      sources.set(field.key, source)
    }
  }
  for (const field of userFields) {
    if (fields.some(item => item.key === field.key)) continue
    fields.push(toMatrixField(schema.type, field, 'user'))
  }
  return {
    type: schema.type, label: schema.label, pluginId: schema.pluginId ?? '',
    traits: [...schema._traitIds], facets: [...(schema.facets ?? [])], fields,
  }
}
```

`toMatrixField(entityType, field, source)` 的职责：把 `source` 映射为 `sourceKind/sourceLabel`；生成 notes（顶层字段、placeholder、defaultValue、autoLink、semanticHint、refType、覆盖说明、`multi-select` 归一、事件 `parentId` 属性树说明）；生成 `search` = entityType + key/label/type/source/notes/options 小写拼接。

- [ ] **Step 3: 实现事件环**

```ts
import { ref } from 'vue'

export type AutoLinkRuntimeEventKind =
  | 'index:start' | 'index:end' | 'link:add'
  | 'link:resolve' | 'decision:record' | 'refresh:target'

export interface AutoLinkRuntimeEvent {
  id: number
  kind: AutoLinkRuntimeEventKind
  at: string
  summary: string
  payload?: Record<string, unknown>
}

const MAX_EVENTS = 200
const events = ref<AutoLinkRuntimeEvent[]>([])
const version = ref(0)
let seq = 0

export function recordAutoLinkEvent(
  kind: AutoLinkRuntimeEventKind,
  summary: string,
  payload?: Record<string, unknown>,
): AutoLinkRuntimeEvent {
  seq += 1
  const event: AutoLinkRuntimeEvent = { id: seq, kind, at: new Date().toISOString(), summary, payload }
  events.value = [event, ...events.value].slice(0, MAX_EVENTS)
  version.value += 1
  return event
}

export function clearAutoLinkEvents(): void {
  events.value = []
  version.value += 1
}

export const autoLinkRuntime = { events, version, clear: clearAutoLinkEvents }
```

```ts
export function useAutoLinkObserver() {
  const version = autoLinkRuntime.version
  const paused = ref(false)
  const frozen = ref<AutoLinkRuntimeEvent[]>([])
  const links = ref(potentialLinkIndex.getAll())
  const decisions = ref(autoLinkDecisionStore.getAll())
  const strategies = computed(() => autoLinkStrategyRegistry.getAll())
  const hints = computed(() => semanticHintRegistry.getAll())
  const mappings = computed(() => userFieldMappingStore.getAll())

  function refresh() {
    links.value = potentialLinkIndex.getAll()
    decisions.value = autoLinkDecisionStore.getAll()
  }
  const stop = watch(version, () => { if (!paused.value) refresh() })
  onBeforeUnmount(stop)
  const events = computed(() => paused.value ? frozen.value : autoLinkRuntime.events.value)
  function togglePause() {
    if (!paused.value) frozen.value = autoLinkRuntime.events.value.slice()
    paused.value = !paused.value
    if (!paused.value) refresh()
  }
  function clearEvents() { autoLinkRuntime.clear() }
  return { events, links, decisions, strategies, hints, mappings, paused, togglePause, clearEvents, refresh }
}
```
