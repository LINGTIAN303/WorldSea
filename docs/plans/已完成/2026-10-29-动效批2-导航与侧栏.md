`apps/tauri/src/ui/layout/ShellSurface.vue` 的 `<script setup>` 改为：

```ts
import { onBeforeUnmount, ref, watch } from 'vue'

const props = withDefaults(defineProps<{
  surface: 'home' | 'workspace'
  inert?: boolean
  busy?: boolean
  workspaceClass?: Record<string, boolean> | string | Array<string | Record<string, boolean>>
}>(), { inert: false, busy: false, workspaceClass: () => ({}) })

/** 工作台常驻、首页覆盖在其上；切到首页时延后隐藏工作台，避免淡入时背后空白。 */
const workspaceShown = ref(props.surface === 'workspace')
let hideTimer: ReturnType<typeof setTimeout> | null = null

function clearHideTimer() {
  if (hideTimer) {
    clearTimeout(hideTimer)
    hideTimer = null
  }
}

watch(
  () => props.surface,
  (surface) => {
    clearHideTimer()
    if (surface === 'workspace') {
      workspaceShown.value = true
      return
    }
    const root = document.documentElement
    const instant = root.dataset.reduceMotion === 'on' || root.dataset.animNav === 'off'
    if (instant) {
      workspaceShown.value = false
      return
    }
    hideTimer = setTimeout(() => {
      workspaceShown.value = false
      hideTimer = null
    }, 180)
  },
)

onBeforeUnmount(clearHideTimer)
```

- [x] **Step 2: 实现 composable**

创建 `packages/ui-kit/src/composables/useViewEntering.ts`：

```ts
import { nextTick, onScopeDispose, ref, watch } from 'vue'

/**
 * 视图切换的入场窗口：视图 id 变化时置 true，duration 后自动回落。
 * 连续切换时先摘再挂，保证动画重新开始。
 */
export function useViewEntering(viewId: () => string | null, duration = 180) {
  const entering = ref(false)
  let timer: ReturnType<typeof setTimeout> | null = null
  let seq = 0

  function start() {
    if (timer) clearTimeout(timer)
    timer = setTimeout(() => {
      entering.value = false
      timer = null
    }, duration)
  }

  watch(viewId, async (next, prev) => {
    if (!next || next === prev) return
    const mine = ++seq
    if (!entering.value) {
      entering.value = true
      start()
      return
    }
    entering.value = false
    await nextTick()
    if (mine !== seq) return
    entering.value = true
    start()
  })

  onScopeDispose(() => {
    seq++
    if (timer) {
      clearTimeout(timer)
      timer = null
    }
  })

  return entering
}
```
