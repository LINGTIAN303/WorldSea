## Global Constraints

- 六目标 key：`canvas` / `home` / `titlebar` / `sidebar` / `content-toolbar` / `content-view`；`canvas` 的 UI 文案改为「大背景」。
- 旧 `content` 配置只迁移到 `content-view`；不复制到 `content-toolbar`。
- 每目标亮/暗两图共用 `enabled/fit/focusX/focusY/dim/blur`；工具条/视图/窗口栏/侧栏的 `fit` 固定 `cover`、焦点固定 `50/50`。

- 工具条目标只在 bar 形态挂载；overlay 胶囊与 none 折叠不套背景。
- 不新增裸色值，使用 `--sf-*` / 现有令牌；`style:check` 不新增失败（预存 `rgba 166/164` 由引入方处理）。
- 模块边界：`style-kit` 不依赖 Tauri；`entity-core` 不新增 `@worldsea/style-kit` 依赖；`ui-kit` 不依赖 `apps`。

归一化旧 `content`：

```ts
/** 旧 content 键：新 content-view 缺失时作为迁移输入；不复制到 content-toolbar */
function legacyContent(raw: Record<string, unknown>): unknown {
  return raw.content
}

export function normalizeGlobalConfig(raw: unknown): BackgroundGlobalConfig {
  const value = raw && typeof raw === 'object' ? raw as Record<string, unknown> : {}
  const targetsRaw = value.targets && typeof value.targets === 'object'
    ? value.targets as Record<string, unknown>
    : {}
  const targets = {} as Record<BackgroundTarget, BackgroundTargetConfig>
  for (const target of BACKGROUND_TARGETS) {
    const entry = targetsRaw[target] ?? (target === 'content-view' ? legacyContent(targetsRaw) : undefined)
    targets[target] = normalizeTargetConfig(entry, target)
  }
  return { version: 2, targets }
}

export function normalizeProjectConfig(raw: unknown): BackgroundProjectConfig | null {
  const value = raw && typeof raw === 'object' ? raw as Record<string, unknown> : {}
  if (!value.targets || typeof value.targets !== 'object') return null
  const targetsRaw = value.targets as Record<string, unknown>
  const targets: Partial<Record<BackgroundTarget, BackgroundTargetConfig>> = {}
  let count = 0
  for (const target of BACKGROUND_TARGETS) {
    const entry = targetsRaw[target] ?? (target === 'content-view' ? legacyContent(targetsRaw) : undefined)
    if (!entry || typeof entry !== 'object') continue
    targets[target] = normalizeTargetConfig(entry, target)
    count++
  }
  return count ? { version: 1, targets } : null
}
```

import { useBackground, type BackgroundTarget } from '@worldsea/style-kit'
import BackgroundLayer from '../../layout/BackgroundLayer.vue'
import { BACKGROUND_TARGET_LABELS, BACKGROUND_WORKSPACE_TARGETS } from './backgroundTargets'

const props = defineProps<{
  scope: 'global' | 'project'
  map: 'workspace' | 'home'
  selected: BackgroundTarget
  disabled: boolean
}>()
const emit = defineEmits<{ select: [BackgroundTarget]; 'switch-map': ['workspace' | 'home'] }>()

const { effective, srcs, missing, projectConfig } = useBackground()
const WORKSPACE_REGIONS = BACKGROUND_WORKSPACE_TARGETS.map((target) => ({
  target,
  label: BACKGROUND_TARGET_LABELS[target],
}))

function stateOf(target: BackgroundTarget): { key: 'follow' | 'missing' | 'unset' | 'on' | 'off'; text: string } {
  const follow = props.scope === 'project' && !projectConfig.value?.targets[target]
  if (follow) return { key: 'follow', text: '跟随' }
  const miss = missing.value[target]
  if (miss.light && miss.dark) return { key: 'missing', text: '缺图' }
  const src = srcs.value[target]
  if (!src.light && !src.dark) return { key: 'unset', text: '未设置' }
  return effective.value[target].enabled
    ? { key: 'on', text: '开' }
    : { key: 'off', text: '关' }
}
