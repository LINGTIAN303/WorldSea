```ts
import { usePluginStore, useUIStore, readLastViewId } from '@worldsea/entity-core'
import { toastWarn } from '@worldsea/ui-kit'
import { resolveDefaultViewId, visibleViewsInSidebarOrder } from '@/ui/layout/sidebar/viewOrder'

export function useHomeNavigation() {
  const ui = useUIStore()
  const plugins = usePluginStore()

  function openHome() {
    ui.openHome()
  }

  function openWorkspace(viewId?: string, entityId?: string): boolean {
    const visible = visibleViewsInSidebarOrder()
    const preferred = viewId && plugins.getView(viewId)
      ? viewId
      : ui.currentView && visible.some(v => v.id === ui.currentView)
        ? ui.currentView
        : resolveDefaultViewId(visible, readLastViewId())
    const view = preferred ? plugins.getView(preferred) : undefined
    if (!preferred || !view) {
      toastWarn('没有可用视图，请先在设置中启用芽')
      return false
    }
    ui.setView(preferred)
    ui.viewComponent = view.component
    if (entityId) {
      ui.selectEntity(entityId)
      window.dispatchEvent(new CustomEvent('ws-select-entity', { detail: { entityId } }))
    }
    ui.setSurface('workspace')
    return true
  }

  return { openHome, openWorkspace }
}

import { defineStore } from 'pinia'
import { ref } from 'vue'
import { invoke } from '@tauri-apps/api/core'
import { getFileStorageBackend, isTauri } from '@worldsea/entity-core/core'

export interface HomeProfile {
  version: 1
  tagline: string
  about: string
  coverImage: string
  updatedAt: string | null
}

export const HOME_PROFILE_LIMITS = { tagline: 80, about: 240 } as const

export function defaultHomeProfile(): HomeProfile {
  return { version: 1, tagline: '', about: '', coverImage: '', updatedAt: null }
}

function asText(value: unknown, max: number): string {
  return typeof value === 'string' ? value.trim().slice(0, max) : ''
}

export function normalizeHomeProfile(value: unknown): HomeProfile {
  if (!value || typeof value !== 'object') return defaultHomeProfile()
  const raw = value as Record<string, unknown>
  return {
    version: 1,
    tagline: asText(raw.tagline, HOME_PROFILE_LIMITS.tagline),
    about: asText(raw.about, HOME_PROFILE_LIMITS.about),
    coverImage: asText(raw.coverImage, 2048),
    updatedAt: typeof raw.updatedAt === 'string' ? raw.updatedAt : null,
  }
}

export function homeProfilePath(projectDir: string): string {
  return `${projectDir.replace(/\\/g, '/')}/.worldsea/home.json`
}

export const useHomeProfileStore = defineStore('homeProfile', () => {
  const profile = ref<HomeProfile>(defaultHomeProfile())
  const loading = ref(false)
  const saving = ref(false)
  const error = ref<string | null>(null)

  async function load(): Promise<void> {
    const dir = getFileStorageBackend()?.projectDir
    if (!isTauri() || !dir) {
      profile.value = defaultHomeProfile()
      return
    }
    loading.value = true
    error.value = null
    try {
      const text = await invoke<string>('cmd_fs_read', { path: homeProfilePath(dir), encoding: 'utf-8' })
      profile.value = normalizeHomeProfile(JSON.parse(text))
    } catch {
      // 文件不存在/损坏：首页用默认值，不把可选文件当用户可见错误
      profile.value = defaultHomeProfile()
    } finally {
      loading.value = false
    }
  }

  async function save(patch: Partial<Pick<HomeProfile, 'tagline' | 'about' | 'coverImage'>>): Promise<boolean> {
    const dir = getFileStorageBackend()?.projectDir
    if (!isTauri() || !dir) {
      error.value = '当前环境不支持保存首页简介'
      return false
    }
    saving.value = true
    error.value = null
    const next = normalizeHomeProfile({ ...profile.value, ...patch, updatedAt: new Date().toISOString() })
    try {
      await invoke('cmd_fs_write', {
        path: homeProfilePath(dir),
        content: JSON.stringify(next, null, 2),
        createDirs: true,
      })
      profile.value = next
      return true
    } catch (e) {
      error.value = e instanceof Error ? e.message : String(e)
      return false
    } finally {
      saving.value = false
    }
  }

  function reset() {
    profile.value = defaultHomeProfile()
    error.value = null
  }

  return { profile, loading, saving, error, load, save, reset }
})

export function relativeTime(iso: string, now = Date.now()): string {
  const then = Date.parse(iso)
  if (!Number.isFinite(then)) return ''
  const diff = Math.max(0, now - then)
  const minute = 60_000
  const hour = 60 * minute
  const day = 24 * hour
  if (diff < minute) return '刚刚'
  if (diff < hour) return `${Math.floor(diff / minute)} 分钟前`
  if (diff < day) return `${Math.floor(diff / hour)} 小时前`
  if (diff < 2 * day) return '昨天'
  const d = new Date(then)
  return `${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`
}

    (max, e) => (!max || e.updatedAt > max ? e.updatedAt : max), null,

    outlineCount: typeCounts.get('outline_node') ?? 0,
    manuscriptCount: typeCounts.get('manuscript') ?? 0,

    : `${snapshot.entityCount} 个实体 · ${snapshot.relationCount} 条关系 · ${snapshot.activeSproutCount} 个芽在运行`

import { computed, onMounted, ref, watch } from 'vue'
import {
  entitySchemaRegistry,
  readLastViewId,
  useEntityStore,
  useRelationStore,
  useSettingsStore,
  useUIStore,
} from '@worldsea/entity-core'
import { availablePlugins } from '@/world'
import { pluginEntityTypes } from '@/ui/layout/sidebar/sproutEntityCounts'
import { visibleViewsInSidebarOrder } from '@/ui/layout/sidebar/viewOrder'
import { useFieldMatrixData } from '@/world/plugins/field-matrix/composables/useFieldMatrixData'
import { useProjectSwitcher } from '@/composables/useProjectSwitcher'
import { useHomeProfileStore } from '@/stores/homeProfileStore'
import { buildHomeData, type HomeData } from './homeData'

export function waitUntilIdle(read: () => boolean): Promise<void> {
  if (!read()) return Promise.resolve()
  return new Promise(resolve => {
    const stop = watch(read, value => {
      if (!value) {
        stop()
        resolve()
      }
    })
  })
}

export function useHomeData() {
  const entityStore = useEntityStore()
  const relationStore = useRelationStore()
  const settingsStore = useSettingsStore()
  const ui = useUIStore()
  const project = useProjectSwitcher()
  const profileStore = useHomeProfileStore()
  const matrix = useFieldMatrixData()
  const bootstrapped = ref(false)
  const loading = computed(() => !bootstrapped.value)
  const error = ref<string | null>(null)

  const projectName = computed(() => {
    const p = project.currentProject.value
    return p?.isDefault ? '默认工作区' : (p?.name || '默认工作区')
  })
  const typeMeta = computed(() => new Map(
    entitySchemaRegistry.getAll().map(s => [s.typeKey, { label: s.label, icon: s.icon ?? 'entity', pluginId: s.pluginId }]),
  ))
  const pluginTypes = computed(() => pluginEntityTypes())

  const data = computed<HomeData | null>(() => {
    if (!bootstrapped.value) return null
    return buildHomeData({
      projectName: projectName.value,
      profile: profileStore.profile,
      entities: entityStore.entities,
      relations: relationStore.relations,
      sprouts: settingsStore.sprouts,
      plugins: availablePlugins(),
      visibleViews: visibleViewsInSidebarOrder(),
      pluginTypes: pluginTypes.value,
      typeMeta: typeMeta.value,
      matrix: matrix.model.value,
      preferredViewId: ui.currentView ?? readLastViewId(),
    })
  })

  async function ensureLoaded() {
    try {
      if (entityStore.loading) await waitUntilIdle(() => entityStore.loading)
      if (!entityStore.entities.length) await entityStore.loadAll()
      if (relationStore.loading) await waitUntilIdle(() => relationStore.loading)
      if (!relationStore.relations.length) await relationStore.loadAll()
      error.value = entityStore.error
    } catch (e) {
      error.value = e instanceof Error ? e.message : String(e)
    } finally {
      bootstrapped.value = true
    }
  }

  async function reload() {
    bootstrapped.value = false
    error.value = null
    await ensureLoaded()
  }

  onMounted(ensureLoaded)
  return { data, loading, error, reload }
}
```

        </div>

              <span v-else class="hp-cover-empty"><WsIcon name="image" size="lg" /></span>
            </div>
            <div class="hp-cover-actions">
              <TuiButton size="sm" variant="secondary" @click="coverInput?.click()">上传封面</TuiButton>
              <TuiButton v-if="draft.coverImage" size="sm" variant="secondary" @click="removeCover">移除封面</TuiButton>
            </div>
            <input ref="coverInput" type="file" accept="image/*" class="hp-hidden" @change="onPickCover" />
          </div>
        </div>

      </div>

  </CardDeck>

<script setup lang="ts">
import { computed, onBeforeUnmount, onMounted, reactive, ref, watch } from 'vue'
import { CardDeck, TuiButton, WsIcon } from '@worldsea/ui-kit'
import { useHomeProfileStore } from '@/stores/homeProfileStore'
import { writeImageAsset } from '@/composables/imageAssets'

defineProps<{ searchQuery?: string }>()
const emit = defineEmits<{ 'card-change': [index: number] }>()

const store = useHomeProfileStore()
const draft = reactive({ tagline: '', about: '', coverImage: '' })
const coverInput = ref<HTMLInputElement | null>(null)
const deckRef = ref<InstanceType<typeof CardDeck> | null>(null)
const saveState = ref<'idle' | 'saving' | 'saved' | 'error'>('idle')
const errorText = ref('')
let timer: number | undefined

watch(() => store.profile, p => {
  draft.tagline = p.tagline
  draft.about = p.about
  draft.coverImage = p.coverImage
}, { immediate: true, deep: true })

const statusText = computed(() => {
  if (saveState.value === 'saving') return '保存中…'
  if (saveState.value === 'error') return `保存失败：${errorText.value}`
  return '已自动保存'
})

function scheduleSave() {
  saveState.value = 'idle'
  if (timer) window.clearTimeout(timer)
  timer = window.setTimeout(() => { void flushSave() }, 400)
}

async function flushSave() {
  if (timer) {
    window.clearTimeout(timer)
    timer = undefined
  }
  saveState.value = 'saving'
  const ok = await store.save({ tagline: draft.tagline, about: draft.about, coverImage: draft.coverImage })
  if (ok) {
    saveState.value = 'saved'
    errorText.value = ''
  } else {
    saveState.value = 'error'
    errorText.value = store.error ?? '未知错误'
  }
}

async function onPickCover(e: Event) {
  const input = e.target as HTMLInputElement
  const file = input.files?.[0]
  input.value = ''
  if (!file) return
  const result = await writeImageAsset(file)
  if (!result.ok || !result.src) {
    saveState.value = 'error'
    errorText.value = result.reason ?? '封面上传失败'
    return
  }
  draft.coverImage = result.src
  await flushSave()
}

async function removeCover() {
  draft.coverImage = ''
  await flushSave()
}

onMounted(() => {
  void store.load()
})

onBeforeUnmount(() => {
  if (timer) {
    window.clearTimeout(timer)
    void flushSave()
  }
})

defineExpose({ goToCard: (i: number) => deckRef.value?.goToCard(i) })
</script>
