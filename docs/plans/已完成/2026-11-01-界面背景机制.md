# 界面背景机制 v1 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 给工作台窗口增加一张用户可配置的全局画布背景图，含 6 项配置，并与现有 UI 风格机制协作。

**Architecture:** 背景层挂在 `Shell` 根部的 `.shell` 内 z0，所有 UI 表面在其上；启用时只透明化 `.shell-surface-root` / `.view-container` 的画布底色，侧栏/工具条/视图画布/首页层保持原表面。配置与根变量放 `style-kit`，全局图片导入放 `entity-core`，背景组件与设置卡放 `ui-kit`，`apps` 只做装配。首页继续使用项目简介封面，不新增首页背景。

**Tech Stack:** Vue 3 + TypeScript、Tauri 2、Vitest + @vue/test-utils、CSS 变量/`backdrop-filter`、现有自绘文件选择器与 `cmd_fs_*` 命令。

**Spec:** `docs/details/架构/界面背景机制.md`

## Global Constraints

- v1 只做工作台画布背景；首页及其项目封面不变。
- 配置 6 项：图片、启用、填充、焦点、暗化、模糊；全局用户级；不按项目、不按主题分图。
- `fit` 仅 `cover | contain | tile`；焦点仅 `0 | 50 | 100` 两轴；`dim` 0–80；`blur` 0–30。
- `data-workspace-bg` 只在 `enabled && src` 时写入；`--ws-bg-*` 在有 `src` 时写入（设置卡预览在关闭态也要显示）。
- 背景层祖先不得加 `isolation` / `filter` / `contrast()`；图片模糊只作用图片元素自身。
- 模块边界：`style-kit` 不依赖 Tauri；`ui-kit` 不得反向依赖 `apps`；`entity-core` 承担 Tauri 资产能力。
- 图片：PNG/JPG/JPEG/WebP，输入 ≤20MB，长边 ≤2560px，落 `<app_data>/worldsea/backgrounds/`。
- 新增样式不得增加 `style:check` 基线；使用 `--sf-*` / 现有令牌；UTF-8 无 BOM；代码标识符英文、注释中文。
- 每任务提交用明确路径 `git add`，禁止 `-A`；不触碰其他 Agent 的未提交文件。

```ts
/** 工作台背景配置：全局 localStorage + 根属性/变量；不依赖 Tauri。 */
import { ref } from 'vue'

export type BackgroundFit = 'cover' | 'contain' | 'tile'
export type FocusStep = 0 | 50 | 100

export interface WorkspaceBackground {
  version: 1
  enabled: boolean
  path: string
  src: string
  fit: BackgroundFit
  focusX: FocusStep
  focusY: FocusStep
  dim: number
  blur: number
}

export const BACKGROUND_KEY = 'worldsea_workspace_background'
export const BACKGROUND_DEFAULTS: WorkspaceBackground = {
  version: 1, enabled: false, path: '', src: '',
  fit: 'cover', focusX: 50, focusY: 50, dim: 35, blur: 8,
}
export const BACKGROUND_VARS = [
  '--ws-bg-image', '--ws-bg-size', '--ws-bg-repeat',
  '--ws-bg-position', '--ws-bg-dim', '--ws-bg-blur',
] as const

const FOCUS_STEPS: FocusStep[] = [0, 50, 100]
const FIT_VALUES: BackgroundFit[] = ['cover', 'contain', 'tile']

function nearestFocus(value: unknown): FocusStep {
  const n = typeof value === 'number' && Number.isFinite(value) ? value : 50
  return FOCUS_STEPS.reduce((best, step) => Math.abs(step - n) < Math.abs(best - n) ? step : best, 50)
}

function clamp(value: unknown, min: number, max: number, fallback: number): number {
  const n = typeof value === 'number' && Number.isFinite(value) ? value : fallback
  return Math.min(max, Math.max(min, Math.round(n)))
}

export function normalizeBackground(raw: unknown): WorkspaceBackground {
  const value = raw && typeof raw === 'object' ? raw as Record<string, unknown> : {}
  const fit = typeof value.fit === 'string' && FIT_VALUES.includes(value.fit as BackgroundFit)
    ? value.fit as BackgroundFit
    : 'cover'
  return {
    version: 1,
    enabled: value.enabled === true,
    path: typeof value.path === 'string' ? value.path : '',
    src: typeof value.src === 'string' ? value.src : '',
    fit,
    focusX: nearestFocus(value.focusX),
    focusY: nearestFocus(value.focusY),
    dim: clamp(value.dim, 0, 80, 35),
    blur: clamp(value.blur, 0, 30, 8),
  }
}

const state = ref<WorkspaceBackground>({ ...BACKGROUND_DEFAULTS })
const missing = ref(false)
let loaded = false
let persistTimer: ReturnType<typeof setTimeout> | null = null

function storage(): Storage | null {
  try { return typeof localStorage === 'undefined' ? null : localStorage } catch { return null }
}

function load() {
  if (loaded) return
  loaded = true
  const raw = storage()?.getItem(BACKGROUND_KEY)
  if (!raw) return
  try { state.value = normalizeBackground(JSON.parse(raw)) } catch { state.value = { ...BACKGROUND_DEFAULTS } }
}

function persistNow() {
  if (persistTimer) { clearTimeout(persistTimer); persistTimer = null }
  storage()?.setItem(BACKGROUND_KEY, JSON.stringify(state.value))
}

function schedulePersist() {
  if (persistTimer) clearTimeout(persistTimer)
  persistTimer = setTimeout(() => { persistTimer = null; persistNow() }, 150)
}

function apply() {
  const root = document.documentElement
  const bg = state.value
  if (bg.src) {
    root.style.setProperty('--ws-bg-image', `url("${bg.src}")`)
    root.style.setProperty('--ws-bg-size', bg.fit === 'cover' ? 'cover' : bg.fit === 'contain' ? 'contain' : 'auto')
    root.style.setProperty('--ws-bg-repeat', bg.fit === 'tile' ? 'repeat' : 'no-repeat')
    root.style.setProperty('--ws-bg-position', `${bg.focusX}% ${bg.focusY}%`)
    root.style.setProperty('--ws-bg-dim', String(bg.dim))
    root.style.setProperty('--ws-bg-blur', `${bg.blur}px`)
  } else {
    for (const name of BACKGROUND_VARS) root.style.removeProperty(name)
  }
  if (bg.enabled && bg.src) root.dataset.workspaceBg = 'on'
  else delete root.dataset.workspaceBg
}

function setEnabled(on: boolean) {
  state.value = { ...state.value, enabled: on }
  apply()
  persistNow()
}
function setImage(image: { path: string; src: string }) {
  state.value = { ...state.value, ...image, enabled: true }
  missing.value = false
  apply()
  persistNow()
}
function setSrc(src: string) {
  state.value = { ...state.value, src }
  missing.value = false
  apply()
  persistNow()
}
function markMissing() {
  state.value = { ...state.value, src: '', enabled: false }
  missing.value = true
  apply()
  persistNow()
}
function clear() {
  state.value = { ...BACKGROUND_DEFAULTS }
  missing.value = false
  apply()
  persistNow()
}
function setFit(fit: BackgroundFit) {
  if (!FIT_VALUES.includes(fit)) return
  state.value = { ...state.value, fit }
  apply()
  schedulePersist()
}
function setFocus(x: FocusStep, y: FocusStep) {
  state.value = { ...state.value, focusX: nearestFocus(x), focusY: nearestFocus(y) }
  apply()
  schedulePersist()
}
function setDim(value: number) {
  state.value = { ...state.value, dim: clamp(value, 0, 80, 35) }
  apply()
  schedulePersist()
}
function setBlur(value: number) {
  state.value = { ...state.value, blur: clamp(value, 0, 30, 8) }
  apply()
  schedulePersist()
}

export function applyPersistedBackground() {
  load()
  apply()
}

export function useBackground() {
  load()
  return { background: state, missing, setEnabled, setImage, setSrc, markMissing, clear, setFit, setFocus, setDim, setBlur, apply, flush: persistNow }
}
```

```ts
/** 工作台全局背景图：选择、长边归一化、写入 app data、删除/解析；不进入项目目录。 */
import { invoke, convertFileSrc } from '@tauri-apps/api/core'
import { appDataDir, join } from '@tauri-apps/api/path'
import { pickLocalFiles } from './pickLocalFiles'
import { isTauri } from './StorageBackend'
import { bytesToBase64 } from './customIcons'

export const BACKGROUND_MAX_BYTES = 20 * 1024 * 1024
export const BACKGROUND_MAX_EDGE = 2560
const ALLOWED_EXTS = new Set(['png', 'jpg', 'jpeg', 'webp'])

export interface BackgroundImportResult {
  ok: boolean
  path?: string
  src?: string
  reason?: string
}

export function fitLongEdge(width: number, height: number, max = BACKGROUND_MAX_EDGE): { width: number; height: number } {
  const w = Math.max(1, Math.round(width))
  const h = Math.max(1, Math.round(height))
  const scale = Math.min(1, max / Math.max(w, h))
  return { width: Math.max(1, Math.round(w * scale)), height: Math.max(1, Math.round(h * scale)) }
}

export function backgroundFileName(ext: string, now = Date.now()): string {
  const safe = (ext || '').toLowerCase().replace(/[^a-z0-9]/g, '')
  return `workspace-${now}.${ALLOWED_EXTS.has(safe) ? safe : 'webp'}`
}

async function encodeBackground(file: File): Promise<{ bytes: Uint8Array; ext: string } | null> {
  if (!file.type.startsWith('image/')) return null
  const bitmap = await createImageBitmap(file)
  try {
    const size = fitLongEdge(bitmap.width, bitmap.height)
    const canvas = document.createElement('canvas')
    canvas.width = size.width
    canvas.height = size.height
    const ctx = canvas.getContext('2d')
    if (!ctx) return null
    ctx.drawImage(bitmap, 0, 0, size.width, size.height)
    const toBlob = (type: string, quality: number) =>
      new Promise<Blob | null>((resolve) => canvas.toBlob(resolve, type, quality))
    let blob = await toBlob('image/webp', 0.9)
    let ext = 'webp'
    if (!blob) {
      blob = await toBlob('image/jpeg', 0.85)
      ext = 'jpg'
    }
    if (!blob) return null
    return { bytes: new Uint8Array(await blob.arrayBuffer()), ext }
  } finally {
    bitmap.close()
  }
}

export async function pickAndSaveWorkspaceBackground(): Promise<BackgroundImportResult | null> {
  if (!isTauri()) return { ok: false, reason: '当前环境不支持选择文件' }
  const picked = await pickLocalFiles({
    multiple: false,
    title: '选择工作台背景',
    filters: [{ name: '图片', extensions: ['png', 'jpg', 'jpeg', 'webp'] }],
    maxBytes: BACKGROUND_MAX_BYTES,
  })
  if (picked.cancelled) return null
  const file = picked.items[0]
  if (!file) return { ok: false, reason: picked.reasons[0] ?? '没有可用的图片' }
  let encoded: { bytes: Uint8Array; ext: string } | null = null
  try {
    encoded = await encodeBackground(file)
  } catch {
    encoded = null
  }
  if (!encoded) return { ok: false, reason: '图片解码失败，请换一张 PNG/JPG/WebP' }
  try {
    const dir = await join(await appDataDir(), 'worldsea', 'backgrounds')
    await invoke('cmd_fs_mkdir', { path: dir })
    const path = await join(dir, backgroundFileName(encoded.ext))
    await invoke('cmd_fs_write', { path, content: bytesToBase64(encoded.bytes), createDirs: true, encoding: 'base64' })
    return { ok: true, path, src: convertFileSrc(path) }
  } catch (err) {
    return { ok: false, reason: err instanceof Error ? err.message : String(err) }
  }
}

export async function removeWorkspaceBackground(path: string): Promise<void> {
  if (!path) return
  try {
    await invoke('cmd_fs_delete', { path })
  } catch (err) {
    console.warn('[background] 删除背景文件失败:', err)
  }
}

export async function resolveWorkspaceBackgroundSrc(path: string): Promise<string | null> {
  if (!path) return null
  try {
    await invoke('cmd_fs_stat', { path })
    return convertFileSrc(path)
  } catch {
    return null
  }
}
```

```vue
import { computed, ref } from 'vue'
import { useBackground, type BackgroundFit, type FocusStep } from '@worldsea/style-kit'
import { pickAndSaveWorkspaceBackground, removeWorkspaceBackground } from '@worldsea/entity-core/core'
import WorkspaceBackground from '../../layout/WorkspaceBackground.vue'

const { background, missing, setEnabled, setImage, clear, setFit, setFocus, setDim, setBlur, flush } = useBackground()
const picking = ref(false)
const error = ref('')
const hasBackground = computed(() => !!background.value.src)
const FOCUS_CELLS: Array<{ key: string; x: FocusStep; y: FocusStep; label: string }> = [
  { key: 'tl', x: 0, y: 0, label: '左上' },
  { key: 'tc', x: 50, y: 0, label: '上中' },
  { key: 'tr', x: 100, y: 0, label: '右上' },
  { key: 'ml', x: 0, y: 50, label: '左中' },
  { key: 'mc', x: 50, y: 50, label: '居中' },
  { key: 'mr', x: 100, y: 50, label: '右中' },
  { key: 'bl', x: 0, y: 100, label: '左下' },
  { key: 'bc', x: 50, y: 100, label: '下中' },
  { key: 'br', x: 100, y: 100, label: '右下' },
]

async function onPick() {
  if (picking.value) return
  picking.value = true
  error.value = ''
  try {
    const oldPath = background.value.path
    const result = await pickAndSaveWorkspaceBackground()
    if (result === null) return
    if (!result.ok || !result.path || !result.src) {
      error.value = result.reason ?? '背景导入失败'
      return
    }
    setImage({ path: result.path, src: result.src })
    if (oldPath && oldPath !== result.path) void removeWorkspaceBackground(oldPath)
  } catch (err) {
    error.value = err instanceof Error ? err.message : String(err)
  } finally {
    picking.value = false
  }
}

function onRemove() {
  const oldPath = background.value.path
  clear()
  error.value = ''
  if (oldPath) void removeWorkspaceBackground(oldPath)
}

function onToggle(on: boolean) {
  setEnabled(on)
}
function onFit(event: Event) {
  setFit((event.target as HTMLSelectElement).value as BackgroundFit)
}
function onFocus(x: FocusStep, y: FocusStep) {
  setFocus(x, y)
}
function onDim(event: Event) {
  setDim(Number((event.target as HTMLInputElement).value))
}
function onBlur(event: Event) {
  setBlur(Number((event.target as HTMLInputElement).value))
}
```
