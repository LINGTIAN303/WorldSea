# UI 组件风格机制 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在产品域实现可切换的三套 UI 组件风格（`mist` / `soft-glass` / `neu`），并以表面契约支撑材质、实体、形变、触感四维微调。

**Architecture:** 新建 `packages/style-kit`，以 13 个表面角色（`--sf-*`）为契约；风格包只提供角色配方，并把现有通用令牌（`--bg-surface` / `--border` / `--shadow-*` / `--glass-*`）重指向角色层，让存量组件零改动跟随。风格选择与四维覆盖持久化在产品域 localStorage，经 `<html data-style>` 生效，不进入壳 `prefs.json`、不影响 dsh。

**Tech Stack:** Vue 3、TypeScript、原生 CSS 变量、Vitest（jsdom）、Vite。

**Spec:** [2026-10-30-UI组件风格机制.md](2026-10-30-UI组件风格机制.md)

## 全局约束

- 仅产品域：不修改 `dsh/`、不改壳 `prefs.json`、不向 dsh 窗广播。
- 不新增第三方依赖；只用现有 Vue / Vitest / Node。
- 所有文件 UTF-8 无 BOM；代码标识符英文、注释中文。
- 代码内只写一行简要注释，详细说明放 `docs/details/注释/packages/style-kit/index.md`。

- 每批走查通过后必须重建产品域产物：
  `cd apps/tauri && node node_modules/vite/bin/vite.js build --config workspace/vite.config.ts`
  判据 = `apps/tauri/workspace-dist/index.html` 时间戳更新。
- 批次门禁：`vue-tsc --noEmit` 0 错误、相关 vitest 全绿、`npm run style:check` 通过。
- 每批完成即停，等用户真机走查通过后再开下一批。

export interface StyleAxes {
  material: number
  depth: number
  morph: number
  tactile: number
}

export interface StylePack {
  id: StyleId
  label: string
  description: string
  axes: StyleAxes
}

export const STYLE_PACKS: Record<StyleId, StylePack> = {
  mist: {
    id: 'mist',
    label: '雾面',
    description: '扁与液态之间：实边框定界、单级阴影、形变克制',

import { describe, it, expect } from 'vitest'
import { readFileSync } from 'node:fs'
import { resolve } from 'node:path'
import { STYLE_PACKS } from '../packs'
import { REQUIRED_VARS, FORBIDDEN_VARS } from '../contract'

const read = (name: string) =>
  readFileSync(resolve(process.cwd(), `packages/style-kit/src/presets/${name}.css`), 'utf8')

describe('预设 CSS 契约', () => {
  it('每套预设定义全部角色变量', () => {
    for (const id of Object.keys(STYLE_PACKS)) {
      const css = read(id)
      for (const v of REQUIRED_VARS) {
        expect(css, `${id} 缺少 ${v}`).toContain(`${v}:`)
      }
    }
  })

  it('预设不得改写文字与语义令牌', () => {
    for (const id of Object.keys(STYLE_PACKS)) {
      const css = read(id)
      for (const v of FORBIDDEN_VARS) {
        expect(css, `${id} 不得定义 ${v}`).not.toContain(`${v}:`)
      }
    }
  })

  it('预设不得引用会被重指向的令牌（防循环）', () => {
    const banned = [
      'var(--bg-app)', 'var(--bg-surface)', 'var(--bg-muted)',
      'var(--border)', 'var(--border-strong)',
      'var(--shadow-sm)', 'var(--shadow-md)', 'var(--shadow-lg)',
      'var(--glass-bg)', 'var(--glass-blur)',
    ]
    for (const id of Object.keys(STYLE_PACKS)) {
      const css = read(id)
      for (const token of banned) {
        expect(css, `${id} 引用了 ${token}`).not.toContain(token)
      }
    }
  })

  it('选择器与风格 id 对应', () => {
    for (const id of Object.keys(STYLE_PACKS)) {
      expect(read(id)).toContain(`html[data-style='${id}']`)
    }
  })
})

```ts
/** 风格状态单例：data-style 由本模块唯一控制，持久化键见下方常量；详细说明见 docs/details/注释/packages/style-kit/index.md */
import { computed, ref } from 'vue'
import { AXIS_META, DEFAULT_STYLE_ID, STYLE_PACKS, type AxisKey, type StyleAxes, type StyleId } from '../packs'

const STYLE_KEY = 'worldsea_ui_style'
const AXES_KEY = 'worldsea_ui_style_axes'
const PERF_KEY = 'worldsea_ui_style_perf'

const styleId = ref<StyleId>(DEFAULT_STYLE_ID)
const overrides = ref<Partial<StyleAxes>>({})
const perfMode = ref(false)
let loaded = false

function storage(): Storage | null {
  try { return typeof localStorage === 'undefined' ? null : localStorage } catch { return null }
}

function isStyleId(value: unknown): value is StyleId {
  return typeof value === 'string' && value in STYLE_PACKS
}

function clampAxis(key: AxisKey, value: number): number {
  const meta = AXIS_META[key]
  return Math.min(meta.max, Math.max(meta.min, Math.round(value)))
}

function load() {
  if (loaded) return
  loaded = true
  const ls = storage()
  if (!ls) return
  const saved = ls.getItem(STYLE_KEY)
  if (isStyleId(saved)) styleId.value = saved
  perfMode.value = ls.getItem(PERF_KEY) === '1'
  try {
    const raw = ls.getItem(AXES_KEY)
    const parsed = raw ? (JSON.parse(raw) as Partial<StyleAxes>) : null
    if (parsed) {
      const next: Partial<StyleAxes> = {}
      for (const key of Object.keys(AXIS_META) as AxisKey[]) {
        const value = parsed[key]
        if (typeof value === 'number' && Number.isFinite(value)) next[key] = clampAxis(key, value)
      }
      overrides.value = next
    }
  } catch {
    overrides.value = {}
  }
}

const resolvedAxes = computed<StyleAxes>(() => ({ ...STYLE_PACKS[styleId.value].axes, ...overrides.value }))

function apply() {
  const root = document.documentElement
  root.dataset.style = styleId.value
  root.dataset.perf = perfMode.value ? 'on' : 'off'
  const axes = resolvedAxes.value
  root.style.setProperty('--sf-material', String(axes.material))
  root.style.setProperty('--sf-depth', String(axes.depth))
  root.style.setProperty('--sf-morph', String(axes.morph))
  root.style.setProperty('--sf-tactile', String(axes.tactile))
}

function persist() {
  const ls = storage()
  if (!ls) return
  ls.setItem(STYLE_KEY, styleId.value)
  ls.setItem(PERF_KEY, perfMode.value ? '1' : '0')
  if (Object.keys(overrides.value).length) ls.setItem(AXES_KEY, JSON.stringify(overrides.value))
  else ls.removeItem(AXES_KEY)
}

export function applyPersistedStyle() {
  load()
  apply()
}

export function useStyle() {
  load()

  function setStyle(id: StyleId) {
    styleId.value = id
    persist()
    apply()
  }

  function setAxis(key: AxisKey, value: number) {
    overrides.value = { ...overrides.value, [key]: clampAxis(key, value) }
    persist()
    apply()
  }

  function resetAxis(key: AxisKey) {
    const next = { ...overrides.value }
    delete next[key]
    overrides.value = next
    persist()
    apply()
  }

  function resetAxes() {
    overrides.value = {}
    persist()
    apply()
  }

  function setPerfMode(on: boolean) {
    perfMode.value = on
    persist()
    apply()
  }

  function isOverridden(key: AxisKey) {
    return overrides.value[key] !== undefined
  }

  return { styleId, overrides, resolvedAxes, perfMode, setStyle, setAxis, resetAxis, resetAxes, setPerfMode, isOverridden }
}
```

`scripts/style-conformance/check.mjs`：

```js
#!/usr/bin/env node
/** 样式一致性门禁：禁止在目标目录的样式块中新增硬编码颜色与阴影；存量以基线锁死。 */
import { existsSync, readFileSync, readdirSync, statSync, writeFileSync } from 'node:fs'
import { join, relative, resolve } from 'node:path'

const ROOT = resolve(process.cwd())
const BASELINE = resolve(ROOT, 'scripts/style-conformance/baseline.json')
const TARGETS = ['packages/ui-kit/src', 'apps/tauri/src']
const WHITELIST = ['packages/ui-kit/src/assets/', 'apps/tauri/src/assets/', '/graph/']
const RULES = {
  hex: /#[0-9a-fA-F]{3,8}\b/g,
  rgba: /rgba?\(/g,
  boxShadow: /box-shadow\s*:/g,
  backdropFilter: /backdrop-filter\s*:/g,
}

function walk(dir, out = []) {
  for (const name of readdirSync(dir)) {
    const full = join(dir, name)
    const stat = statSync(full)
    if (stat.isDirectory()) {
      if (['node_modules', 'dist', 'workspace-dist', '__tests__', 'assets'].includes(name)) continue
      walk(full, out)
    } else if (name.endsWith('.vue') || name.endsWith('.css')) {
      out.push(full)
    }
  }
  return out
}

function styleSource(file, source) {
  if (file.endsWith('.css')) return source
  return [...source.matchAll(/<style[^>]*>([\s\S]*?)<\/style>/g)].map((m) => m[1]).join('\n')
}

const counts = { hex: 0, rgba: 0, boxShadow: 0, backdropFilter: 0 }
for (const target of TARGETS) {
  for (const file of walk(resolve(ROOT, target))) {
    const rel = relative(ROOT, file).replace(/\\/g, '/')
    if (WHITELIST.some((prefix) => `/${rel}`.includes(prefix))) continue
    const css = styleSource(file, readFileSync(file, 'utf8'))
    for (const [key, re] of Object.entries(RULES)) counts[key] += (css.match(re) ?? []).length
  }
}

if (process.argv.includes('--write-baseline')) {
  writeFileSync(BASELINE, `${JSON.stringify(counts, null, 2)}\n`, 'utf8')
  console.log('style-conformance baseline ->', counts)
  process.exit(0)
}

if (!existsSync(BASELINE)) {
  console.error('缺少基线：先运行 node scripts/style-conformance/check.mjs --write-baseline')
  process.exit(1)
}

const baseline = JSON.parse(readFileSync(BASELINE, 'utf8'))
let failed = false
for (const key of Object.keys(RULES)) {
  const current = counts[key]
  if (current > baseline[key]) {
    console.error(`✗ ${key}: ${current} > 基线 ${baseline[key]}`)
    failed = true
  } else {
    console.log(`✓ ${key}: ${current} / ${baseline[key]}`)
  }
}
process.exit(failed ? 1 : 0)
```
